(function () {
  var root = document.getElementById('ssb-bonus-slider');
  if (!root) return;

  var viewport = root.querySelector('.ssb-bonus__viewport');
  var slides = Array.prototype.slice.call(root.querySelectorAll('.ssb-bonus__slide'));
  var dotsWrap = root.querySelector('.ssb-bonus__dots');
  var nextBtn = root.querySelector('.ssb-bonus__arrow--next');
  var visible = 1;
  var pages = 1;
  var pageWidth = 0;
  var currentPage = 0;

  function getVisible() {
    if (window.innerWidth <= 767) return 1;
    if (window.innerWidth <= 1080) return 3;
    return 4;
  }

  function buildDots() {
    dotsWrap.innerHTML = '';
    for (var i = 0; i < pages; i++) {
      var dot = document.createElement('button');
      dot.type = 'button';
      dot.className = 'ssb-bonus__dot';
      dot.setAttribute('aria-label', 'Ir al slide ' + (i + 1));
      dot.addEventListener('click', (function (page) {
        return function () {
          scrollToPage(page);
        };
      })(i));
      dotsWrap.appendChild(dot);
    }
  }

  function updateDots() {
    var index = pageWidth ? Math.round(viewport.scrollLeft / pageWidth) : 0;
    if (index < 0) index = 0;
    if (index > pages - 1) index = pages - 1;
    currentPage = index;
    var dots = dotsWrap.querySelectorAll('.ssb-bonus__dot');
    for (var i = 0; i < dots.length; i++) {
      dots[i].classList.toggle('is-active', i === index);
    }
  }

  function scrollToPage(page) {
    if (page < 0) page = 0;
    if (page > pages - 1) page = pages - 1;
    currentPage = page;
    viewport.scrollTo({
      left: pageWidth * page,
      behavior: 'smooth'
    });
  }

  function getGap() {
    return window.innerWidth <= 767 ? 12 : 16;
  }

  function recalc() {
    visible = getVisible();
    var slideWidth = slides[0] ? slides[0].getBoundingClientRect().width : 0;
    pageWidth = (slideWidth * visible) + (getGap() * Math.max(0, visible - 1));
    pages = Math.max(1, Math.ceil(slides.length / visible));
    buildDots();
    updateDots();
    if (nextBtn) {
      nextBtn.disabled = pages <= 1;
    }
  }

  viewport.addEventListener('scroll', updateDots);

  if (nextBtn) {
    nextBtn.addEventListener('click', function () {
      var nextPage = currentPage + 1;
      if (nextPage >= pages) nextPage = 0;
      scrollToPage(nextPage);
    });
  }

  window.addEventListener('resize', recalc);
  recalc();
})();
</script>
